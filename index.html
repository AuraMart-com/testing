# Export current security policy
secedit /export /cfg C:\secpol.cfg

# Disable complexity requirement in the config file
(Get-Content C:\secpol.cfg) -replace 'PasswordComplexity = 1', 'PasswordComplexity = 0' | Set-Content C:\secpol.cfg

# Apply the updated policy
secedit /configure /db $env:windir\security\local.sdb /cfg C:\secpol.cfg /areas SECURITYPOLICY

# Clean up temporary config file
Remove-Item C:\secpol.cfg
